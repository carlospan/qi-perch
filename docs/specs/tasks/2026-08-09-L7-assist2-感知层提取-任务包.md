### 1. `qi/action/volition.py` 加 `parse_assist_request`

```python
import re
from dataclasses import dataclass

@dataclass(frozen=True)
class AssistRequest:
    op: str          # "read_file"
    target_path: str  # 规范化后的路径

# 文件路径正则：Windows 绝对 / POSIX 绝对 / ~/Home / 相对含分隔符
_PATH_PATTERNS = [
    re.compile(r"([A-Za-z]:[\\/][^\s'""，。、]+)"),      # D:\... 或 D:/...
    re.compile(r"(/(?:[^\s'""，。、/]+/)*[^\s'""，。、/]+)"),  # /abs/path
    re.compile(r"(~[\\/][^\s'""，。、]+)"),                # ~/diary.md
    re.compile(r"([\w./-]+\.txt|\.md|\.json|\.csv|\.log|\.py|\.js|\.ts)", re.I),  # note.txt / diary.md（相对）
]

_OP_CUES = {
    "read_file": ("读", "看", "打开", "瞧", "瞄", "查"),
}

def parse_assist_request(message: str | None) -> AssistRequest | None:
    """从用户消息提取协助请求。无路径返回 None。"""
    text = (message or "").strip()
    if not text:
        return None
    # op 提取（本包只 read_file）
    op = None
    for cue in _OP_CUES["read_file"]:
        if cue in text:
            op = "read_file"
            break
    if op is None:
        return None
    # 路径提取
    for pattern in _PATH_PATTERNS:
        m = pattern.search(text)
        if m:
            path = m.group(1).strip("""'""，。、")
            # 规范化：反斜杠统一
            path = path.replace("\\", "/")
            return AssistRequest(op=op, target_path=path)
    return None
```

> 注：本包只做 `read_file` op；其他 op（write_file 等）留后续。正则保守——宁可漏（无路径 → None → assist 不执行）不可错（误读非用户文件）。

import pytest
from qi.action.volition import parse_assist_request, AssistRequest


class TestParseAssistRequest:
    def test_windows_absolute_path(self):
        r = parse_assist_request("帮我看一下 D:/notes/today.txt")
        assert r is not None
        assert r.op == "read_file"
        assert "notes/today.txt" in r.target_path

    def test_windows_backslash_path(self):
        r = parse_assist_request("读一下 D:\\diary\\today.md")
        assert r is not None
        assert r.target_path == "D:/diary/today.md"

    def test_posix_absolute_path(self):
        r = parse_assist_request("帮我看 /tmp/log.txt")
        assert r is not None
        assert r.target_path == "/tmp/log.txt"

    def test_home_path(self):
        r = parse_assist_request("帮我读 ~/diary.md")
        assert r is not None
        assert r.target_path == "~/diary.md"

    def test_relative_path_with_ext(self):
        r = parse_assist_request("看下 note.txt")
        assert r is not None
        assert r.target_path == "note.txt"

    def test_no_path_returns_none(self):
        """有「看」但无路径 → None（assist.execute 返回 None）。"""
        r = parse_assist_request("帮我看看这段")
        assert r is None

        """有路径但无「读/看」→ None。"""

        assert parse_assist_request("") is None

    def test_path_with_trailing_punctuation(self):
        """路径尾标点剥离。"""
