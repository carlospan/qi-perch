### 1. `qi/action/volition.py`

**新增**（L52 附近，`looks_like_help_request` 后）：

```python
import re
from dataclasses import dataclass

@dataclass(frozen=True)
class AssistRequest:
    op: str
    target_path: str

_PATH_PATTERNS = [
    re.compile(r"([A-Za-z]:[\\/][^\s'""，。、]+)"),
    re.compile(r"(/(?:[^\s'""，。、/]+/)*[^\s'""，。、/]+)"),
    re.compile(r"(~[\\/][^\s'""，。、]+)"),
    re.compile(r"([\w./-]+\.txt|\.md|\.json|\.csv|\.log|\.py|\.js|\.ts)", re.I),
]

_OP_CUES_READ = ("读", "看", "打开", "瞧", "瞄", "查")

def parse_assist_request(message: str | None) -> AssistRequest | None:
    """从用户消息提取协助请求。无路径或无 op cue 返回 None。"""
    text = (message or "").strip()
    if not text:
        return None
    if not any(cue in text for cue in _OP_CUES_READ):
        return None
    for pattern in _PATH_PATTERNS:
        m = pattern.search(text)
        if m:
            path = m.group(1).strip("""'""，。、")
            path = path.replace("\\", "/")
            return AssistRequest(op="read_file", target_path=path)
    return None
```

**export**：`__init__.py` `__all__` 加 `"parse_assist_request"` + `"AssistRequest"` + import。
