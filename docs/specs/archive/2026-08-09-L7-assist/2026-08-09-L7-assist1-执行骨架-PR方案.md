1. `execute_kind(kind="assist", op="read_file", target_path=..., confirmed=...)` → 栖执行读文件 + LLM 复述
2. friend+ 已确认 → 读文件 → digest → success + speak

## 精确改动点

### 1. `qi/action/assist.py`（新建）

完整骨架见任务包「实现步骤概要 §1」。关键点：

- `AssistAction.execute(op, target_path, *, relationship_stage, trust, scars, confirmed, season, now)` → dict
- permission 检查：`can_read_user_file(relationship_stage, trust, scars)` → `(allowed, needs_confirm)`
- 确认门：`needs_confirm and not confirmed` → `_confirm_gate`（不执行 + qi_line 请求确认 + outcome="confirm_required"）
- 读文件：`Path(target_path).expanduser().resolve().read_text(...)[:_MAX_READ_BYTES]`
- digest：`_digest_file` LLM consciousness 复述（失败降级只说文件名）
- 失败：`_fail(qi_line, summary, outcome)` 统一格式

**常量**：
- `_MAX_READ_BYTES = 32768`（32KB）
- `_PRIVACY_LINE = "不外传用户文件内容原文"`

```python
"""assist-1：读用户文件 + 确认门 + LLM 复述。"""

import pytest
from datetime import datetime
from qi.action.assist import AssistAction
from qi.action.layer import ActionLayer
from qi.core.emotion import EmotionState


@pytest.mark.asyncio
async def test_assist_read_file_success(db, tmp_path):
    """friend+ 已确认 → 读文件 → digest → success。"""
    f = tmp_path / "note.txt"
    f.write_text("今天天气不错", encoding="utf-8")
    assist = AssistAction(db, llm=_FakeLLM("看到了今天的天气记一笔。"))
    result = await assist.execute(
        "read_file", str(f),
        relationship_stage="friend",
        trust=0.7,
        confirmed=True,
        season="spring",
        now=datetime(2026, 8, 9),
    )
    assert result["outcome"] == "success"
    assert result["speak"] is True
    assert "天气" in result["qi_line"]  # digest 含内容

@pytest.mark.asyncio
async def test_assist_not_confirmed_returns_confirm_gate(db, tmp_path):
    """friend+ 未确认 → confirm_gate，不执行。"""
    f = tmp_path / "note.txt"
    f.write_text("secret", encoding="utf-8")
    assist = AssistAction(db, llm=_FakeLLM("不该走到这"))
    result = await assist.execute(
        "read_file", str(f),
        relationship_stage="friend",
        trust=0.7,
        confirmed=False,  # 未确认
        season="spring",
        now=datetime(2026, 8, 9),
    )
    assert result["outcome"] == "confirm_required"
    assert "说一声" in result["qi_line"]
    assert result.get("needs_confirmation") is True

@pytest.mark.asyncio
async def test_assist_stranger_not_allowed(db, tmp_path):
    """stranger → 不允许，failed_capability。"""
    f = tmp_path / "note.txt"
    f.write_text("x", encoding="utf-8")
    assist = AssistAction(db, llm=_FakeLLM("不该"))
    result = await assist.execute(
        "read_file", str(f),
        relationship_stage="stranger",
        confirmed=True,
        season="spring",
        now=datetime(2026, 8, 9),
    )
    assert result["outcome"] == "failed_capability"
    assert "熟一点" in result["qi_line"]

@pytest.mark.asyncio
async def test_assist_file_not_found(db, tmp_path):
    """文件不存在 → failed_capability。"""
    assist = AssistAction(db, llm=_FakeLLM("不该"))
    result = await assist.execute(
        "read_file", str(tmp_path / "nope.txt"),

        confirmed=True,
        season="spring",
        now=datetime(2026, 8, 9),
    )
    assert result["outcome"] == "failed_capability"
    assert "没找到" in result["qi_line"]

@pytest.mark.asyncio
async def test_assist_scar_blocks(db, tmp_path):
    """assist 伤疤 → 不允许。"""
    # 先造伤疤
    await db.save_scar(
        origin_event="[action:assist] 之前越界了",

    scars = await db.list_scars()

    result = await assist.execute(
        "read_file", str(f),
        relationship_stage="friend",
        trust=0.7,
        scars=scars,
        confirmed=True,
        season="spring",
        now=datetime(2026, 8, 9),
    )
    assert result["outcome"] == "failed_capability"  # scar_blocks_kind 拦

@pytest.mark.asyncio
async def test_assist_digest_uses_llm(db, tmp_path):
    """digest 走 LLM consciousness。"""

@pytest.mark.asyncio
async def test_execute_kind_assist_not_record_budget(db, tmp_path):
    """execute_kind 走 assist 不调 budget.record。"""
    f = tmp_path / "note.txt"
    f.write_text("x", encoding="utf-8")
    layer = ActionLayer(db, {"action": {"autonomous_daily_limit": 20}}, llm=_FakeLLM("看了。"))
    before = layer.budget.count_today
    await layer.execute_kind(
        "assist",
        EmotionState(curiosity=0.5),
        "friend",
        "spring",
        datetime(2026, 8, 9),
        mode="solitary",
        op="read_file",
        target_path=str(f),
        confirmed=True,
    )
    assert layer.budget.count_today == before  # 不占预算


class _FakeLLM:
    def __init__(self, text: str):
        self.text = text
        self.calls = []
    async def call(self, *, purpose, messages):
        self.calls.append({"purpose": purpose, "messages": messages})
        return self.text
```
