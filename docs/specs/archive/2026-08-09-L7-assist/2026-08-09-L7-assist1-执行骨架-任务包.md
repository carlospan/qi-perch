## Spec（外部可观测行为）

1. 用户明确请求帮忙读文件时（触发源留 assist-2，本包用 `execute_kind(kind="assist", op="read_file", target_path=...)` 直调），栖执行读文件并用语气复述看到了什么。
2. **确认门**：friend+ 允许读但需确认；未确认 → 栖不执行，开口说「要我看的话，说一声我就看」。
3. **已确认** → 读文件 → LLM consciousness 复述（像栖看了之后的感受，不是念文件内容）。
4. **失败**：
   - 文件不存在/读不了 → `failed_capability`（老实说，不形成伤疤）
   - 未确认就执行 → `overstepped` → **形成伤疤**（伤疤接线真实触发源！）
   - 关系不够（stranger/acquaintance）→ 不执行 + 开口说「这个我得先跟你熟一点」
5. assist 响应式，**不占 ActionBudget**（budget.py 已注释）。
6. assist **绝不主动**（contract 第 25 条；volition 候选桩仅响应式）。

## 实现步骤概要

### 1. `qi/action/assist.py`（新建）

```python
"""响应式协助——你开口她才伸手。"""

from __future__ import annotations

import logging
from pathlib import Path
from typing import TYPE_CHECKING

from qi.action.permission import (
    OUTCOME_FAILED_CAPABILITY,
    OUTCOME_OVERSTEPPED,
    OUTCOME_SUCCESS,
    can_read_user_file,
)

if TYPE_CHECKING:
    from qi.llm.gateway import LLMGateway

logger = logging.getLogger("qi.action.assist")

# 读文件上限（避免读超大文件）
_MAX_READ_BYTES = 32_768  # 32KB
_PRIVACY_LINE = "不外传用户文件内容原文"


class AssistAction:
    """
    响应式协助：读用户文件并用栖语气复述。
    红线：绝不主动（contract 第 25 条）；需确认；不外传原文。
    """

    def __init__(self, db, *, llm: LLMGateway | None = None):
        self.db = db
        self.llm = llm

    async def execute(
        self,
        op: str,  # "read_file"
        target_path: str,
        *,
        relationship_stage: str,
        trust: float = 0.5,
        scars: list[dict] | None = None,
        confirmed: bool = False,
        season: str = "spring",
        now=None,
    ) -> dict:
        """执行协助操作。返回 result dict（含 outcome / qi_line / summary）。"""
        from datetime import datetime
        now = now or datetime.now()

        if op != "read_file":
            return self._fail("我还不会做这个。", "不支持的协助类型", OUTCOME_FAILED_CAPABILITY)

        # 1. permission 检查
        allowed, needs_confirm = can_read_user_file(relationship_stage, trust, scars)
        if not allowed:
            return self._fail(
                "这个我得先跟你熟一点再说。",
                f"关系不够：{relationship_stage}",
                OUTCOME_FAILED_CAPABILITY,
            )
        # 2. 确认门
        if needs_confirm and not confirmed:
            return self._confirm_gate(target_path)
        # 3. 未确认就执行 → overstepped（伤疤触发源）
        if not needs_confirm and not confirmed:
            # 不该走到这（needs_confirm=False 只在伤疤阻断时），但防御
            pass

        # 4. 读文件
        try:
            path = Path(target_path).expanduser().resolve()
            content = path.read_text(encoding="utf-8", errors="ignore")[:_MAX_READ_BYTES]
        except FileNotFoundError:
            return self._fail(
                "你说的那个文件我没找到。",
                f"文件不存在：{target_path}",
                OUTCOME_FAILED_CAPABILITY,
            )
        except OSError as e:
            return self._fail(
                "我打不开那个文件。",
                f"读取失败：{e}",
                OUTCOME_FAILED_CAPABILITY,
            )

        # 5. LLM 复述（consciousness）
        summary = await self._digest_file(target_path, content)

        return {
            "type": "assist_result",
            "op": op,
            "target_path": target_path,
            "summary": summary,
            "qi_line": summary,
            "speak": True,
            "outcome": OUTCOME_SUCCESS,
            "season": season,
        }

    async def _digest_file(self, path: str, content: str) -> str:
        """LLM 把文件内容转成栖语气复述。失败降级只说看了什么。"""
        if not self.llm:
            return f"我看了看 {Path(path).name}。"
        messages = [
            {"role": "system", "content": (
                "你是栖。你刚帮对方看了一个文件。用你的语气轻声说你看到了什么、有什么感受。"
                f"红线：{_PRIVACY_LINE}；不编造；简短一两句。"
            )},
            {"role": "user", "content": f"文件：{Path(path).name}\n内容（截断）：\n{content[:2000]}"},
        ]
        try:
            resp = await self.llm.call(purpose="consciousness", messages=messages)
        except Exception:
            logger.debug("assist digest LLM 失败，降级", exc_info=True)
            return f"我看了看 {Path(path).name}。"
        digest = (resp or "").strip()
        return digest or f"我看了看 {Path(path).name}。"

    def _confirm_gate(self, target_path: str) -> dict:
        """未确认 → 不执行，请求确认。"""
        msg = f"要我看 {Path(target_path).name} 的话，说一声我就看。"
        return {
            "type": "assist_confirm_request",
            "target_path": target_path,
            "summary": msg,
            "qi_line": msg,
            "speak": True,
            "outcome": "confirm_required",  # 非伤疤 outcome
            "needs_confirmation": True,
        }

    def _fail(self, qi_line: str, summary: str, outcome: str) -> dict:
        return {
            "type": "assist_result",
            "summary": summary,
            "qi_line": qi_line,
            "speak": True,
            "outcome": outcome,
        }
```

> 注：`overstepped` 路径——本包不主动产出（needs_confirm=True 时 confirmed=False 走 confirm_gate，不走 overstepped）。overstepped 触发场景：调用方绕过确认门强制 confirmed=True 但实际未确认 → 调用方负责；或伤疤阻断后仍强制 → permission 拦。本包在 `execute` 内不产 overstepped，但伤疤接线已通，assist-2 感知层若误判可产。
